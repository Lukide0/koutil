517 template <comptime_t Data> bool remove(const key_t& key, adapter_t adapter) {
518 if (m_buckets_count == 0) {
519 return false;
520 }
521
522 auto hash = hash_of<Data>(key);
523 auto& bucket = m_buckets[index_of_hash(hash)];
524 auto it = find_bucket_item<Data>(key, hash, bucket, adapter);
525
526 if (it == bucket.end()) {
527 return false;
528 }

530 // special case

594 void rehash(std::size_t count) {
595 auto* buckets = create_buckets(count);
596
597 for (std::size_t i = 0; i < m_buckets_count; ++i) {
598 for (auto&& [hash, key_index] : m_buckets[i]) {
599 buckets[hash % count].emplace_back(hash, key_index);
600 }
601 }
602
603 if (m_buckets != nullptr) {
604 std::destroy_n(m_buckets, m_buckets_count);
605 m_alloc.deallocate(m_buckets, m_buckets_count);
606 }
607
608 m_buckets_count = count;
609 m_buckets = buckets;
610 }
